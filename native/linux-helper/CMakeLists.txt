cmake_minimum_required(VERSION 3.20)
project(codevo_input_helper LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(DESKTOP REQUIRED IMPORTED_TARGET gtk+-3.0 json-glib-1.0 atspi-2 libevdev libudev xi xtst x11 xfixes)
add_executable(codevo-input-helper main.cpp server.cpp desktop.cpp selection.cpp keys.cpp)
set_target_properties(codevo-input-helper PROPERTIES SKIP_BUILD_RPATH TRUE)
target_link_libraries(codevo-input-helper PRIVATE PkgConfig::DESKTOP)
target_compile_options(codevo-input-helper PRIVATE -Wall -Wextra -Wpedantic)
